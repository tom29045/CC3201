	nombre varchar (255),
	voto_electoral smallint,
	cierre time,
	num_candidatos smallint
);

INSERT INTO eleccion.GGGGG_estado SELECT * FROM eleccion.estado;

	nombre varchar (255),
	estado varchar (255),
	reportado float
);

INSERT INTO eleccion.GGGGG_condado SELECT * FROM eleccion.condado;

	nombre varchar (255),
	partido varchar (255)
);

INSERT INTO eleccion.GGGGG_candidato SELECT * FROM eleccion.candidato;
